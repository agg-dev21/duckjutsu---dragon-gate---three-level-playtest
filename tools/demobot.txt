
// ===== DEMO BOT (only in the recording copy, not the real game) =====
function tap (b: controller.Button, ms: number) {
    b.setPressed(true)
    pause(ms)
    b.setPressed(false)
}
function run (b: controller.Button, ms: number, throwing: boolean) {
    b.setPressed(true)
    let t = 0
    while (t < ms) {
        if (duck.isHittingTile(CollisionDirection.Right) || duck.isHittingTile(CollisionDirection.Left)) {
            tap(controller.A, 40)
        }
        if (level == 1 && t % 650 == 0) {
            tap(controller.A, 40)
        }
        if (throwing && t % 400 == 0) {
            tap(controller.B, 30)
        }
        if (info.life() < 3) {
            info.setLife(5)
        }
        pause(50)
        t += 50
    }
    b.setPressed(false)
}
function faceAndThrow (target: Sprite) {
    if (target.x > duck.x) {
        tap(controller.right, 40)
    } else {
        tap(controller.left, 40)
    }
    tap(controller.B, 40)
    if (info.life() < 3) {
        info.setLife(5)
    }
}
